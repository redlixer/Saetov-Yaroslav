create table client (
id serial primary key,
last_name varchar(100),
first_name varchar(100),
email varchar (100)
);

create table order_ (
id serial primary key,
client_id serial,
deadline date,
delivery_address varchar(100),
status varchar(100)
);

create table product(
id serial primary key,
order_id serial,
title varchar(100),
size_ varchar(100),
price decimal,
finish varchar(100)
);


create table operation (
id serial primary key,
product_id serial,
title varchar(100),
seq_number int,
planned_date date,
actual_date date
);

create table brigade (
id serial primary key,
brigade_number int,
specialization varchar(100)
);

create table material (
id serial primary key,
product_id serial,
title varchar(100),
unit varchar(100),
planed_qty decimal,
actual_qty decimal
);

create table employee (
id serial primary key,
last_name varchar(100),
first_name varchar(100),
position varchar(100)
);

create table quality_check (
id serial primary key,
product_id serial,
employee_id serial,
check_date date,
result varchar(100)
);
